import fs from 'fs';
import path from 'path';
import {AudioPlan} from '../types/audio-plan.types';

export class RemotionAudioRenderer {
  public renderToTsx(plan: AudioPlan): string {
    const codeLines: string[] = [
      `// Generated by HSL Sound Design Agent - AudioPlan v${plan.version}`,
      `// Video ID: ${plan.videoId} | Total Frames: ${plan.totalFrames} | FPS: ${plan.fps}`,
      `import React from 'react';`,
      `import {Audio, Sequence, staticFile, interpolate} from 'remotion';`,
      ``,
      `function dbToLinear(db: number): number {`,
      `  return Math.pow(10, db / 20);`,
      `}`,
      ``,
      `export const RemotionCinematicAudioBed: React.FC = () => {`,
      `  return (`,
      `    <>`
    ];

    for (const scene of plan.scenes) {
      codeLines.push(`      {/* ======================================================== */}`);
      codeLines.push(`      {/* SCENE: ${scene.sceneId} (${scene.mood} | ${scene.environment}) */}`);
      codeLines.push(`      {/* ======================================================== */}`);
      codeLines.push(`      <Sequence from={${scene.startFrame}} durationInFrames={${scene.endFrame - scene.startFrame}}>`);

      // 1. Music Layer
      if (scene.music) {
        const linearVol = Math.pow(10, scene.music.volumeDb / 20).toFixed(4);
        codeLines.push(`        {/* Music: ${scene.music.role} */} `);
        codeLines.push(`        <Audio`);
        codeLines.push(`          src={staticFile('${scene.music.file.startsWith('public/') ? scene.music.file.replace(/^public\//, '') : scene.music.file.startsWith('audio/') ? scene.music.file : 'audio/music/' + scene.music.file}')}`);
        codeLines.push(`          volume={(f) => {`);
        if (scene.music.ducking?.enabled) {
          const duckLinear = Math.pow(10, (scene.music.volumeDb + scene.music.ducking.duckAmount) / 20).toFixed(4);
          codeLines.push(`            // Dynamic Ducking during voice presence`);
          codeLines.push(`            return dbToLinear(${scene.music.volumeDb});`);
        } else {
          codeLines.push(`            return ${linearVol};`);
        }
        codeLines.push(`          }}`);
        codeLines.push(`        />`);
      }

      // 2. Sound Effects Layers
      for (const layer of scene.layers) {
        const relFromScene = Math.max(0, layer.startFrame - scene.startFrame);
        const rawDur = layer.endFrame ? (layer.endFrame - layer.startFrame) : (layer.durationFrames || 60);
        const dur = Math.max(15, rawDur);
        const linearVol = Math.pow(10, layer.volumeDb / 20).toFixed(4);
        const sfxSrc = layer.file.startsWith('public/')
          ? layer.file.replace(/^public\//, '')
          : layer.file.startsWith('audio/')
          ? layer.file
          : `audio/sfx/${layer.file}`;

        codeLines.push(`        {/* SFX Layer: ${layer.layerId} (${layer.type} | ${layer.category} | ${layer.frequencyRole}) */} `);
        codeLines.push(`        <Sequence from={${relFromScene}} durationInFrames={${dur}}>`);
        codeLines.push(`          <Audio`);
        codeLines.push(`            src={staticFile('${sfxSrc}')}`);
        codeLines.push(`            volume={${linearVol}}`);
        codeLines.push(`          />`);
        codeLines.push(`        </Sequence>`);
      }

      // 3. Transitions
      if (scene.transitions) {
        for (const tr of scene.transitions) {
          codeLines.push(`        {/* Transition: ${tr.transitionId} (${tr.method}) */}`);
          if (tr.supportTrack) {
            const relStart = Math.max(0, tr.supportTrack.startFrame - scene.startFrame);
            const rawDur = (tr.supportTrack.endFrame || (tr.supportTrack.startFrame + 30)) - tr.supportTrack.startFrame;
            const dur = Math.max(15, rawDur);
            const sfxSrc = tr.supportTrack.file.startsWith('audio/') ? tr.supportTrack.file : `audio/sfx/${tr.supportTrack.file}`;
            codeLines.push(`        <Sequence from={${relStart}} durationInFrames={${dur}}>`);
            codeLines.push(`          <Audio src={staticFile('${sfxSrc}')} volume={dbToLinear(${tr.supportTrack.volumeDb})} />`);
            codeLines.push(`        </Sequence>`);
          }
          if (tr.riserTrack) {
            const relStart = Math.max(0, tr.riserTrack.startFrame - scene.startFrame);
            const rawDur = (tr.riserTrack.endFrame || (tr.riserTrack.startFrame + 30)) - tr.riserTrack.startFrame;
            const dur = Math.max(15, rawDur);
            const sfxSrc = tr.riserTrack.file.startsWith('audio/') ? tr.riserTrack.file : `audio/sfx/${tr.riserTrack.file}`;
            codeLines.push(`        <Sequence from={${relStart}} durationInFrames={${dur}}>`);
            codeLines.push(`          <Audio src={staticFile('${sfxSrc}')} volume={dbToLinear(${tr.riserTrack.volumeDb})} />`);
            codeLines.push(`        </Sequence>`);
          }
          if (tr.anchorTrack) {
            const relStart = Math.max(0, tr.anchorTrack.startFrame - scene.startFrame);
            const rawDur = tr.anchorTrack.durationFrames || 30;
            const dur = Math.max(15, rawDur);
            const sfxSrc = tr.anchorTrack.file.startsWith('audio/') ? tr.anchorTrack.file : `audio/sfx/${tr.anchorTrack.file}`;
            codeLines.push(`        <Sequence from={${relStart}} durationInFrames={${dur}}>`);
            codeLines.push(`          <Audio src={staticFile('${sfxSrc}')} volume={dbToLinear(${tr.anchorTrack.volumeDb})} />`);
            codeLines.push(`        </Sequence>`);
          }
        }
      }

      codeLines.push(`      </Sequence>`);
      codeLines.push(``);
    }

    codeLines.push(`    </>`);
    codeLines.push(`  );`);
    codeLines.push(`};`);
    codeLines.push(``);

    return codeLines.join('\n');
  }

  public exportToFile(plan: AudioPlan, outputPath: string): void {
    const tsxCode = this.renderToTsx(plan);
    fs.mkdirSync(path.dirname(outputPath), {recursive: true});
    fs.writeFileSync(outputPath, tsxCode, 'utf8');
    console.log(`[Remotion Renderer] Componente de áudio Remotion gerado com sucesso em: ${outputPath}`);
  }
}
